# 对于 ω₂, N₂=6, 系数为

φ₁(x) = 1      φ₃(x) = 2x₂
φ₂(x) = 2x₁      φ₄(x) = 4x₁x₂

C₂₁ = 1/6 Σₖ₌₁⁶ φ₁( x⃗ₖ⁽²⁾ ) = 1

C₂₂ = 1/6 Σₖ₌₁⁶ φ₂( x⃗ₖ⁽²⁾ ) = 1/6 Σₖ₌₁⁶ 2xₖ₁⁽²⁾ = -2/6 (2+3+3+3+3+4) = -6

C₂₃ = 1/6 Σₖ₌₁⁶ φ₃( x⃗ₖ⁽²⁾ ) = 1/6 Σₖ₌₁⁶ 2xₖ₂⁽²⁾ = -2/6 (3+2+3+4+5+3) = -6.7

C₂₄ = 1/6 Σₖ₌₁⁶ φ₄( x⃗ₖ⁽²⁾ ) = 1/6 Σₖ₌₁⁶ 4xₖ₁⁽²⁾xₖ₂⁽²⁾
= 4/6 (2×3 + 3×2 + 3×3 + 3×4 + 3×5 + 4×3) = 40

∴ p̂_N₂( x⃗ | ω₂ ) = Σⱼ₌₁⁴ C₂ⱼ φⱼ( x⃗ ) = 1 - 6×(2x₁) + 6.7×(2x₂) + 40×(4x₁x₂)
= 1 - 12x₁ -13.4x₂ +160x₁x₂
