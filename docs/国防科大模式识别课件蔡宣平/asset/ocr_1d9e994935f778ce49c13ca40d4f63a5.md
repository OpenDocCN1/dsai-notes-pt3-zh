#include <math.h>
#include <stdlib.h>
void gauss(double * x1,double *x2)
{//产生服从正态分布的随机数对，用变量参数x1与x2传递随机数
    double pi,r1,r2,s;
    pi=3.14159265;      //赋 π 值
    r1=rand()/RAND_MAX ;
    r2=rand()/RAND_MAX ;
    s=sqrt(-2*log(r1));      //中间变量赋值
    *x1=s*cos(2*pi*r2);
    *x2=s*sin(2*pi*r2);
    //正态分布随机数赋给变量参数
}
