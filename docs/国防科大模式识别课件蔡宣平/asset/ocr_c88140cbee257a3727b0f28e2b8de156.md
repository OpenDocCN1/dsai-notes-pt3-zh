# 令k = 1/2，求得准则函数的梯度

∇J(w̄) = ∂J/∂w̄ = 1/2 [x̄ sgn(w̄' x̄) - x̄]

其中符号函数 sgn(w̄' x̄) = { 1      w̄' x̄ > 0
                               -1     w̄' x̄ ≤ 0 }

# 增广权矢量的修正迭代公式为：

w̄(k+1) = w̄(k) - ρ_k ∇J(w̄(k))

= w̄(k) - ρ_k/2 [x̄_k sgn[w̄'(k) x̄_k] - x̄_k]

= { w̄(k)                     若 w̄'(k) x̄_k > 0
    w̄(k) + ρ_k x̄_k           若 w̄'(k) x̄_k ≤ 0 , ρ_k > 0 }
