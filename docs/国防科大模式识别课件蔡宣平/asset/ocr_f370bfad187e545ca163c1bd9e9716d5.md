d(ar{x})=d^r(ar{x})+d^{r-1}(ar{x})+\cdots+d^1(ar{x})+d^0(ar{x}) 	riangleq d^{(r)}(ar{x})

d^0(ar{x}) = w_{n+1} = d^{(0)}(ar{x})

d^1(ar{x}) = \sum_{i=1}^n w_i x_i

d^2(ar{x}) = \sum_{i_1=1}^n \sum_{i_2=i_1}^n w_{i_1 i_2} x_{i_1} x_{i_2}

d^r(ar{x}) = \sum_{i_1=1}^n \cdots \sum_{i_r=i_{r-1}}^n w_{i_1 \dots i_r} x_{i_1} \cdots x_{i_r}
