∂ ln P_h(D) / ∂ w_{ijk} = ∂ / ∂ w_{ijk} ln ∏_{d∈D} P_h(d)
= Σ_{d∈D} ∂ ln P_h(d) / ∂ w_{ijk}
= Σ_{d∈D} 1 / P_h(d) ∂ P_h(d) / ∂ w_{ijk}
= Σ_{d∈D} 1 / P_h(d) ∂ / ∂ w_{ijk} Σ_{j',k'} P_h(d | y_{ij'}, u_{ik'}) P_h(y_{ij'}, u_{ik'})
= Σ_{d∈D} 1 / P_h(d) ∂ / ∂ w_{ijk} Σ_{j',k'} P_h(d | y_{ij'}, u_{ik'}) P_h(y_{ij'} | u_{ik'}) P_h(u_{ik'})
= Σ_{d∈D} 1 / P_h(d) ∂ / ∂ w_{ijk} P_h(d | y_{ij}, u_{ik}) P_h(y_{ij} | u_{ik}) P_h(u_{ik})
= Σ_{d∈D} 1 / P_h(d) ∂ / ∂ w_{ijk} P_h(d | y_{ij}, u_{ik}) w_{ijk} P_h(u_{ik})
= Σ_{d∈D} 1 / P_h(d) P_h(d | y_{ij}, u_{ik}) P_h(u_{ik})
= Σ_{d∈D} 1 / P_h(d) [P_h(y_{ij}, u_{ik} | d) P_h(d) P_h(u_{ik})] / P_h(y_{ij}, u_{ik})
= Σ_{d∈D} P_h(y_{ij}, u_{ik} | d) P_h(u_{ik}) / P_h(y_{ij}, u_{ik})
= Σ_{d∈D} P_h(y_{ij}, u_{ik} | d) / P_h(y_{ij} | u_{ik})
= Σ_{d∈D} P_h(y_{ij}, u_{ik} | d) / w_{ijk}
